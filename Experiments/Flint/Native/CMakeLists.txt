cmake_minimum_required(VERSION 3.24)
project(FlashFlint LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(flag COMMON TESTS EXAMPLES TOOLS SERVER APP UI MTMD)
  set(LLAMA_BUILD_${flag} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(flag OPENMP NATIVE CPU_KLEIDIAI LLAMAFILE ACCELERATE)
  set(GGML_${flag} OFF CACHE BOOL "" FORCE)
endforeach()
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
if(APPLE)
  enable_language(OBJC)
  set(GGML_METAL ON CACHE BOOL "" FORCE)
  set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
else()
  set(GGML_METAL OFF CACHE BOOL "" FORCE)
endif()
if(NOT EXISTS "${OMNI_SOURCE}/flash-local-flint-source.json")
  message(FATAL_ERROR "Run scripts/prepare_flint_runtime.py first")
endif()
add_subdirectory("${OMNI_SOURCE}" omni)
set(LES_NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../../OrnithLES/Native")
target_include_directories(llama PRIVATE "${LES_NATIVE}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/FlintTemplate.jinja")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/FlintTemplate.jinja" FLINT_CHAT_TEMPLATE)
configure_file(FlintTemplate.h.in FlintTemplate.h @ONLY)
add_library(FlashFlint SHARED FlashFlint.cpp FlashFlint.h FlintPrompt.cpp)
target_include_directories(FlashFlint PRIVATE "${OMNI_SOURCE}/src" "${OMNI_SOURCE}/vendor" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(FlashFlint PRIVATE llama)
if(APPLE)
  target_sources(FlashFlint PRIVATE "${LES_NATIVE}/MetalTelemetry.m")
  target_link_libraries(FlashFlint PRIVATE "-framework Foundation" "-framework Metal")
  target_link_options(FlashFlint PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/exports.txt")
  set_target_properties(FlashFlint PROPERTIES FRAMEWORK TRUE FRAMEWORK_VERSION A
    MACOSX_FRAMEWORK_IDENTIFIER com.dhafen.flashlocal.flint
    MACOSX_FRAMEWORK_BUNDLE_VERSION 1 MACOSX_FRAMEWORK_SHORT_VERSION_STRING 1.0.0
    PUBLIC_HEADER FlashFlint.h BUILD_WITH_INSTALL_RPATH TRUE INSTALL_NAME_DIR "@rpath"
    XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO)
else()
  target_link_options(FlashFlint PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
add_executable(flint-control control.cpp)
target_include_directories(flint-control PRIVATE "${OMNI_SOURCE}/src" "${OMNI_SOURCE}/vendor" "${LES_NATIVE}")
target_link_libraries(flint-control PRIVATE llama)
add_executable(flint-quantize-control quantize-control.cpp)
target_include_directories(flint-quantize-control PRIVATE "${OMNI_SOURCE}/vendor")
target_link_libraries(flint-quantize-control PRIVATE llama)
add_executable(flint-api-control api-control.cpp)
target_link_libraries(flint-api-control PRIVATE FlashFlint)
add_executable(flint-vocab-control vocab-control.cpp)
target_include_directories(flint-vocab-control PRIVATE "${OMNI_SOURCE}/vendor")
target_link_libraries(flint-vocab-control PRIVATE llama)
add_executable(flint-prompt-control prompt-control.cpp)
target_link_libraries(flint-prompt-control PRIVATE FlashFlint)
if(APPLE)
  target_sources(flint-control PRIVATE "${LES_NATIVE}/MetalTelemetry.m")
  target_link_libraries(flint-control PRIVATE "-framework Foundation" "-framework Metal")
  target_sources(flint-quantize-control PRIVATE "${LES_NATIVE}/MetalTelemetry.m")
  target_link_libraries(flint-quantize-control PRIVATE "-framework Foundation" "-framework Metal")
  target_sources(flint-vocab-control PRIVATE "${LES_NATIVE}/MetalTelemetry.m")
  target_link_libraries(flint-vocab-control PRIVATE "-framework Foundation" "-framework Metal")
endif()
endif()
